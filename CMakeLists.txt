cmake_minimum_required(VERSION 3.25)
project(radio-field-kit-zero VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(RFK_DEVICE "Build for the Linux ARM64 Cardputer Zero" OFF)
option(RFK_UI "Build the LVGL user interface" ON)
set(RFK_JSON_SOURCE "" CACHE PATH "Path to a pre-fetched nlohmann JSON checkout")
set(RFK_LVGL_SOURCE "" CACHE PATH "Path to a pre-fetched checkout of pinned LVGL")

add_library(field-core src/field.cpp)
target_include_directories(field-core PUBLIC src)
target_compile_options(field-core PRIVATE -Wall -Wextra -Wpedantic)
add_library(rfk-gnss src/gnss.cpp)
target_include_directories(rfk-gnss PUBLIC src)
target_compile_definitions(rfk-gnss PRIVATE RFK_DEVICE=$<BOOL:${RFK_DEVICE}>)
target_compile_options(rfk-gnss PRIVATE -Wall -Wextra -Wpedantic)
add_library(rfk-antennas src/antennas.cpp)
target_include_directories(rfk-antennas PUBLIC src)
target_compile_options(rfk-antennas PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(field-tests tests/field.cpp)
    target_link_libraries(field-tests PRIVATE field-core)
    add_test(NAME field-core COMMAND field-tests)
    add_executable(gnss-tests tests/gnss.cpp)
    target_link_libraries(gnss-tests PRIVATE rfk-gnss)
    add_test(NAME gnss-core COMMAND gnss-tests)
    add_executable(antennas-tests tests/antennas.cpp)
    target_link_libraries(antennas-tests PRIVATE rfk-antennas field-core)
    add_test(NAME antennas COMMAND antennas-tests)
endif()

if(RFK_UI)
    if(RFK_DEVICE AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "RFK_DEVICE requires an ARM64 Linux target.")
    endif()
    include(FetchContent)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
    set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE FILEPATH "" FORCE)
    if(RFK_LVGL_SOURCE)
        set(FETCHCONTENT_SOURCE_DIR_LVGL "${RFK_LVGL_SOURCE}")
    endif()
    FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git
        GIT_TAG 85aa60d18b3d5e5588d7b247abf90198f07c8a63)
    FetchContent_MakeAvailable(lvgl)
    target_compile_definitions(lvgl PUBLIC RFK_DEVICE=$<BOOL:${RFK_DEVICE}>)

    set(JSON_BuildTests OFF CACHE INTERNAL "")
    if(RFK_JSON_SOURCE)
        set(FETCHCONTENT_SOURCE_DIR_NLOHMANN_JSON "${RFK_JSON_SOURCE}")
    endif()
    FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git
        GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
    FetchContent_MakeAvailable(nlohmann_json)
    find_package(Threads REQUIRED)
    add_library(rfk-weather src/weather.cpp)
    target_include_directories(rfk-weather PUBLIC src)
    target_compile_definitions(rfk-weather PRIVATE RFK_DEVICE=$<BOOL:${RFK_DEVICE}>)
    target_compile_options(rfk-weather PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(rfk-weather PRIVATE nlohmann_json::nlohmann_json)
    add_executable(radiofieldkitzero src/main.cpp)
    target_link_libraries(radiofieldkitzero PRIVATE field-core rfk-antennas rfk-gnss rfk-weather Threads::Threads lvgl::lvgl ${CMAKE_DL_LIBS} m)
    target_compile_options(radiofieldkitzero PRIVATE -Wall -Wextra -Wpedantic)
    if(RFK_DEVICE)
        include(cmake/cardputer-package.cmake)
    else()
        find_package(SDL2 CONFIG REQUIRED)
        target_link_libraries(radiofieldkitzero PRIVATE SDL2::SDL2)
        if(BUILD_TESTING)
            add_executable(weather-tests tests/weather.cpp)
            target_link_libraries(weather-tests PRIVATE rfk-weather)
            add_test(NAME weather-utc COMMAND weather-tests)
            add_test(NAME weather-atlantic COMMAND weather-tests)
            set_tests_properties(weather-utc PROPERTIES ENVIRONMENT "TZ=UTC;LC_ALL=C" TIMEOUT 10)
            set_tests_properties(weather-atlantic PROPERTIES ENVIRONMENT "TZ=America/Halifax;LC_ALL=C" TIMEOUT 10)
            add_test(NAME app-smoke COMMAND radiofieldkitzero --smoke)
            set_tests_properties(app-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 10)
        endif()
    endif()
endif()
